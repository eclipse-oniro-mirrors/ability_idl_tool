ErrCode InnerInfoBlockMarshalling(OHOS::MessageParcel& data, const InnerInfo& dataBlock)
{
    InnerInfo dataBlockCp = dataBlock;
    {
        unsigned char flagNorm = *reinterpret_cast<const unsigned char *>(&dataBlockCp.flag);
        dataBlockCp.flag = (flagNorm == 1);
    }
    if (!data.WriteUnpadBuffer(&dataBlockCp, sizeof(InnerInfo))) {
        return ERR_INVALID_DATA;
    }
    return ERR_NONE;
}

ErrCode InnerInfoBlockUnmarshalling(OHOS::MessageParcel& data, InnerInfo& dataBlock)
{
    const InnerInfo *dataBlockPtr = reinterpret_cast<const InnerInfo *>(data.ReadUnpadBuffer(sizeof(InnerInfo)));
    if (dataBlockPtr == nullptr) {
        return ERR_INVALID_DATA;
    }

    if (memcpy_s(&dataBlock, sizeof(InnerInfo), dataBlockPtr, sizeof(InnerInfo)) != EOK) {
        return ERR_INVALID_DATA;
    }

    {
        unsigned char flagNorm = *reinterpret_cast<const unsigned char *>(&dataBlock.flag);
        dataBlock.flag = (flagNorm == 1);
    }
    return ERR_NONE;
}
