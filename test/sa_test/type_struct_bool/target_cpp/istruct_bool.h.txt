#ifndef OHOS_IDL_STRUCTBOOL_ISTRUCTBOOL_H
#define OHOS_IDL_STRUCTBOOL_ISTRUCTBOOL_H

#include <message_parcel.h>
#include <securec.h>
#include <string_ex.h>

namespace ohos {
namespace idl {
namespace structbool {

struct InnerInfo {
    bool flag;
    int32_t id;
};

struct OuterInfo {
    int32_t id;
    InnerInfo inner;
};

struct PlainInfo {
    int32_t id;
};

ErrCode InnerInfoBlockMarshalling(OHOS::MessageParcel &data, const InnerInfo& dataBlock);

ErrCode InnerInfoBlockUnmarshalling(OHOS::MessageParcel &data, InnerInfo& dataBlock);

ErrCode OuterInfoBlockMarshalling(OHOS::MessageParcel &data, const OuterInfo& dataBlock);

ErrCode OuterInfoBlockUnmarshalling(OHOS::MessageParcel &data, OuterInfo& dataBlock);

ErrCode PlainInfoBlockMarshalling(OHOS::MessageParcel &data, const PlainInfo& dataBlock);

ErrCode PlainInfoBlockUnmarshalling(OHOS::MessageParcel &data, PlainInfo& dataBlock);

} // namespace structbool
} // namespace idl
} // namespace ohos

#endif // OHOS_IDL_STRUCTBOOL_ISTRUCTBOOL_H
